182:           party = await isar.partys.filter().partyNameEqualTo(creditNote.partyName!).findFirst();

256:   void _recalculateTotals() {
257:     double sub = 0.0;
258:     double tax = 0.0;
259: 
260:     for (var item in _draftItems) {
261:       final double qty = item.quantity ?? 0.0;
262:       final double rate = item.rate ?? 0.0;
263:       final double itemDisc = item.discount ?? 0.0;
264:       
265:       final lineSub = qty * rate;
266:       final lineTaxable = lineSub - itemDisc;
267:       final lineTax = lineTaxable * ((item.gstRate ?? 0.0) / 100.0);
268:       
269:       item.taxableAmount = lineTaxable;
270:       item.gstAmount = lineTax;
271:       item.totalAmount = lineTaxable + lineTax;
272: 
273:       sub += lineSub;
274:       tax += lineTax;
275:     }
276: 
277:     final discInput = double.tryParse(_discountController.text) ?? 0.0;
278:     _discountAmount = _isDiscountPercent ? (sub * (discInput / 100.0)) : discInput;
279:     final double rawTotal = (sub - _discountAmount) + tax;
280:     _roundOff = _customRoundOff ?? (rawTotal.roundToDouble() - rawTotal);
281: 
282:     setState(() {
283:       _subtotal = sub;
284:       _taxableAmount = sub - _discountAmount;
285:       _totalGST = tax;
286:       _grandTotal = rawTotal + _roundOff;
287:     });
288:   }
